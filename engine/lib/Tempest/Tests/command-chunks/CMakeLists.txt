cmake_minimum_required(VERSION 3.16)
project(TempestCommandChunkTests LANGUAGES CXX)

enable_testing()
add_executable(small-list-tests smalllist.cpp)
target_compile_features(small-list-tests PRIVATE cxx_std_17)
add_test(NAME small-list-reuse COMMAND small-list-tests)

option(TEMPEST_TEST_GPU "Test command-buffer reuse and traversal on a GPU" ON)
if(TEMPEST_TEST_GPU)
  add_executable(command-chunk-tests gpu.cpp)
  target_compile_features(command-chunk-tests PRIVATE cxx_std_17)
  target_include_directories(command-chunk-tests PRIVATE ../../Engine/include)
  set(TEMPEST_TEST_LIBRARY "" CACHE FILEPATH "Optional prebuilt shared Tempest import library")
  if(TEMPEST_TEST_LIBRARY)
    target_link_libraries(command-chunk-tests PRIVATE "${TEMPEST_TEST_LIBRARY}")
  else()
    add_subdirectory(../../Engine tempest)
    target_link_libraries(command-chunk-tests PRIVATE Tempest)
  endif()
  add_test(NAME command-chunks-vulkan COMMAND command-chunk-tests)
  if(WIN32)
    add_test(NAME command-chunks-directx COMMAND command-chunk-tests --directx)
  endif()
endif()

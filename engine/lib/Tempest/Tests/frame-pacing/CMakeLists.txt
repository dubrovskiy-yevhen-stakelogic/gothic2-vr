cmake_minimum_required(VERSION 3.16)
project(FramePacingTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(frame-pacing-tests main.cpp ../../Engine/utility/framepacer.cpp ../../Engine/utility/cputrace.cpp)
target_compile_features(frame-pacing-tests PRIVATE cxx_std_20)
target_compile_definitions(frame-pacing-tests PRIVATE NOMINMAX)
target_include_directories(frame-pacing-tests PRIVATE ../../Engine/include)
target_link_libraries(frame-pacing-tests PRIVATE Threads::Threads)
if(ANDROID)
  target_link_libraries(frame-pacing-tests PRIVATE android)
endif()
add_test(NAME frame-pacing-deadlines-and-wait COMMAND frame-pacing-tests)
set_tests_properties(frame-pacing-deadlines-and-wait PROPERTIES TIMEOUT 30)

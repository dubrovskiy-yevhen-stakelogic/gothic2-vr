cmake_minimum_required(VERSION 3.16)
project(TempestIndexedIndirectTests LANGUAGES CXX)
enable_testing()
add_executable(indexed-indirect-tests main.cpp)
target_compile_features(indexed-indirect-tests PRIVATE cxx_std_20)
target_include_directories(indexed-indirect-tests PRIVATE ../../Engine/include)
set(TEMPEST_TEST_LIBRARY "" CACHE FILEPATH "Prebuilt shared Tempest import library")
if(TEMPEST_TEST_LIBRARY)
  target_link_libraries(indexed-indirect-tests PRIVATE "${TEMPEST_TEST_LIBRARY}")
else()
  add_subdirectory(../../Engine tempest)
  target_link_libraries(indexed-indirect-tests PRIVATE Tempest)
endif()
find_program(GLSLC_EXECUTABLE NAMES glslc REQUIRED)
foreach(stage vert frag comp)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/test.${stage}.spv"
    COMMAND "${GLSLC_EXECUTABLE}" --target-env=vulkan1.0 -O "${CMAKE_CURRENT_SOURCE_DIR}/test.${stage}" -o "${CMAKE_CURRENT_BINARY_DIR}/test.${stage}.spv"
    DEPENDS "test.${stage}" VERBATIM)
  target_sources(indexed-indirect-tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/test.${stage}.spv")
endforeach()
add_test(NAME indexed-indirect-vulkan COMMAND indexed-indirect-tests)
set_tests_properties(indexed-indirect-vulkan PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" SKIP_RETURN_CODE 77)
